def wrap_angle(theta: float) -> float:
    """Wrap to (-pi, pi]."""
    return (theta + math.pi) % (2.0 * math.pi) - math.pi

def edge_key(i: int, j: int):
    return (i, j) if i < j else (j, i)

def oriented_theta(G: nx.Graph, i: int, j: int) -> float:
    """Return oriented theta for traversal i -> j (edges stored undirected with canonical key)."""
    a, b = edge_key(i, j)
    th = G[a][b]["theta"]
    return th if (i == a and j == b) else -th

def U_ij(G: nx.Graph, i: int, j: int) -> complex:
    return np.exp(1j * oriented_theta(G, i, j))

def fit_front_scaling(Rs, ts):
    ts = np.asarray(ts, float)
    Rs = np.asarray(Rs, float)
    if len(ts) < 5:
        return {}

    # Linear: R = a t + b
    A1 = np.vstack([ts, np.ones_like(ts)]).T
    a1, b1 = np.linalg.lstsq(A1, Rs, rcond=None)[0]
    pred1 = a1*ts + b1
    sse1 = float(np.sum((Rs - pred1)**2))

    # Diffusive: R = a sqrt(t) + b
    st = np.sqrt(ts)
    A2 = np.vstack([st, np.ones_like(st)]).T
    a2, b2 = np.linalg.lstsq(A2, Rs, rcond=None)[0]
    pred2 = a2*st + b2
    sse2 = float(np.sum((Rs - pred2)**2))

    return {"lin_a": float(a1), "lin_b": float(b1), "lin_sse": sse1,
            "sqrt_a": float(a2), "sqrt_b": float(b2), "sqrt_sse": sse2}

# ACTION TERMS (LOCAL)
# -----------------------
def self_energy_node(G: nx.Graph, i: int) -> float:
    psi = G.nodes[i]["psi"]
    mag2 = (psi.real * psi.real + psi.imag * psi.imag)
    return MASS2 * mag2 + LAMBDA_PSI * (mag2 * mag2)

def hop_energy_edge(G: nx.Graph, i: int, j: int) -> float:
    """S_hop(i,j) = -KAPPA * Re( conj(psi_i) * U_ij * psi_j )."""
    psi_i = G.nodes[i]["psi"]
    psi_j = G.nodes[j]["psi"]
    val = np.conjugate(psi_i) * U_ij(G, i, j) * psi_j
    return -KAPPA * float(np.real(val))

def triangle_energy(G: nx.Graph, i: int, j: int, k: int) -> float:
    """
    For triangle i-j-k-i:
    S_tri = BETA * (1 - Re(U_ij U_jk U_ki))
    """
    loop = U_ij(G, i, j) * U_ij(G, j, k) * U_ij(G, k, i)
    return BETA * (1.0 - float(np.real(loop)))

def deg2_energy_node(G: nx.Graph, i: int) -> float:
    d = G.degree(i)
    return MU_DEG2 * float(d * d)

def stress_energy_edge(G: nx.Graph, i: int, j: int) -> float:
    """
    Stress–geometry coupling on each existing edge:
    S_stress(i,j) = LAMBDA_G * (|psi_i|^2 + |psi_j|^2)
    """
    rho_i = abs(G.nodes[i]["psi"]) ** 2
    rho_j = abs(G.nodes[j]["psi"]) ** 2
    return LAMBDA_G * float(rho_i + rho_j)

def local_energy_around_node_for_psi(G: nx.Graph, i: int) -> float:
    """Energy terms that depend on psi_i: self(i) + hops + stress on incident edges."""
    e = self_energy_node(G, i)
    for j in G.neighbors(i):
        e += hop_energy_edge(G, i, j)
        # stress term lives on edges, depends on |psi|, so psi updates must include it locally
        e += stress_energy_edge(G, i, j)
    return e

def local_energy_around_edge_for_theta(G: nx.Graph, i: int, j: int) -> float:
    """Energy terms that depend on theta_(i,j): hop(i,j) + triangles using edge(i,j)."""
    if not G.has_edge(*edge_key(i, j)):
        return 0.0
    e = hop_energy_edge(G, i, j)
    # triangles: common neighbors
    common = set(G.neighbors(i)).intersection(set(G.neighbors(j)))
    for k in common:
        e += triangle_energy(G, i, j, k)
    return e

def local_energy_for_rewire(G: nx.Graph, i: int, j: int) -> float:
    """
    Energy pieces affected by edge existence between i and j:
    - hop(i,j) (if exists)
    - stress(i,j) (if exists)
    - triangles containing (i,j)
    - edge count term: LAMBDA_E * |E|
    - deg2 congestion at endpoints
    """
    e = 0.0

    # endpoint deg2 always changes when edge toggles
    e += deg2_energy_node(G, i) + deg2_energy_node(G, j)

    # global edge count term (we include it here because we evaluate before/after)
    e += LAMBDA_E * float(G.number_of_edges())

    if G.has_edge(*edge_key(i, j)):
        e += hop_energy_edge(G, i, j)
        e += stress_energy_edge(G, i, j)
        common = set(G.neighbors(i)).intersection(set(G.neighbors(j)))
        for k in common:
            e += triangle_energy(G, i, j, k)
    return e

def propose_link_target(G: nx.Graph, i: int) -> int | None:
    """Prefer local targets within HORIZON hops; sometimes long-range."""
    if random.random() < P_LONG_RANGE:
        candidates = [v for v in G.nodes() if v != i and not G.has_edge(*edge_key(i, v))]
        return random.choice(candidates) if candidates else None

    visited = {i}
    frontier = {i}
    for _ in range(HORIZON):
        new_frontier = set()
        for u in frontier:
            for v in G.neighbors(u):
                if v not in visited:
                    visited.add(v)
                    new_frontier.add(v)
        frontier = new_frontier

    candidates = [v for v in visited if v != i and not G.has_edge(*edge_key(i, v))]
    return random.choice(candidates) if candidates else None

def sample_triangles(G: nx.Graph, max_tri=1200):
    """Sample triangle triplets (i,j,k) with i<j<k."""
    triangles = []
    nodes = list(G.nodes())
    random.shuffle(nodes)
    for i in nodes:
        Ni = set(G.neighbors(i))
        if len(Ni) < 2:
            continue
        nbrs = list(Ni)
        random.shuffle(nbrs)
        for idx_a in range(min(len(nbrs), 12)):
            a = nbrs[idx_a]
            Na = set(G.neighbors(a))
            common = Na.intersection(Ni)
            for k in list(common)[:6]:
                if k == i or k == a:
                    continue
                tri = tuple(sorted((i, a, k)))
                triangles.append(tri)
                if len(triangles) >= max_tri:
                    return list(set(triangles))
    return list(set(triangles))

def triangle_flux_real(G: nx.Graph, tri):
    """Return Re(U_ij U_jk U_ki) for triangle; nan if triangle is dead."""
    i, j, k = tri
    if not (G.has_edge(*edge_key(i, j)) and G.has_edge(*edge_key(j, k)) and G.has_edge(*edge_key(k, i))):
        return np.nan
    loop = U_ij(G, i, j) * U_ij(G, j, k) * U_ij(G, k, i)
    return float(np.real(loop))

def wilson_line_transporter(G: nx.Graph, path):
    prod = 1.0 + 0.0j
    for a, b in zip(path[:-1], path[1:]):
        prod *= U_ij(G, a, b)
    return prod

def gauge_invariant_two_point(G: nx.Graph, i: int, j: int) -> complex:
    """
    Gauge-invariant correlator using shortest path transporter, normalized to be ~[-1,1]:
      (conj(psi_i) * U_path * psi_j) / (|psi_i||psi_j| + eps)
    """
    psi_i = G.nodes[i]["psi"]
    psi_j = G.nodes[j]["psi"]
    denom = (abs(psi_i) * abs(psi_j) + 1e-12)

    if i == j:
        return (np.conjugate(psi_i) * psi_i) / denom

    try:
        path = nx.shortest_path(G, i, j)
    except nx.NetworkXNoPath:
        return 0.0 + 0.0j

    Upath = wilson_line_transporter(G, path)
    return (np.conjugate(psi_i) * Upath * psi_j) / denom

def apply_random_gauge_transform(G: nx.Graph, rng: np.random.Generator):
    """Apply ψ_i -> e^{iα_i} ψ_i and θ_ab -> θ_ab + α_a - α_b (wrapped)."""
    alpha = {i: float(rng.uniform(-math.pi, math.pi)) for i in G.nodes()}
    for i in G.nodes():
        G.nodes[i]["psi"] *= np.exp(1j * alpha[i])
    for a, b in G.edges():
        th = G[a][b]["theta"]
        G[a][b]["theta"] = wrap_angle(th + alpha[a] - alpha[b])

def correlator_vs_distance(G: nx.Graph, samples=2000, max_d=10, seed=0):
    rng = np.random.default_rng(seed)
    nodes = list(G.nodes())
    bins = defaultdict(list)

    for _ in range(samples):
        i, j = rng.choice(nodes, size=2, replace=False)
        try:
            d = nx.shortest_path_length(G, i, j)
        except nx.NetworkXNoPath:
            continue
        if d > max_d:
            continue
        val = gauge_invariant_two_point(G, i, j)
        bins[d].append(float(np.real(val)))

    ds = list(range(max_d + 1))
    means, counts = [], []
    for d in ds:
        arr = bins.get(d, [])
        counts.append(len(arr))
        means.append(float(np.mean(arr)) if arr else np.nan)
    return np.array(ds), np.array(means), np.array(counts)

def bfs_distances(G: nx.Graph, source: int, max_d: int):
    dist = {source: 0}
    q = deque([source])
    while q:
        u = q.popleft()
        if dist[u] >= max_d:
            continue
        for v in G.neighbors(u):
            if v not in dist:
                dist[v] = dist[u] + 1
                if dist[v] <= max_d:
                    q.append(v)
    return dist

def lightcone_measurement(dist_map, baseline_abspsi, G: nx.Graph, max_d: int):
    """L[d] = mean(|psi|-baseline) over nodes at hop distance d."""
    buckets = [[] for _ in range(max_d + 1)]
    for v, d in dist_map.items():
        if d <= max_d:
            buckets[d].append(abs(G.nodes[v]["psi"]) - baseline_abspsi[v])
    out = np.zeros(max_d + 1, dtype=float)
    for d in range(max_d + 1):
        out[d] = float(np.mean(buckets[d])) if buckets[d] else 0.0
    return out

# CURVATURE PROXY (FORMAN-LIKE)
# -----------------------
def forman_edge_curvature(G: nx.Graph, u: int, v: int) -> float:
    """Cheap proxy: F(u,v)=4-deg(u)-deg(v)."""
    return 4.0 - float(G.degree(u)) - float(G.degree(v))

def node_curvature_proxy(G: nx.Graph, node: int) -> float:
    vals = [forman_edge_curvature(G, node, nbr) for nbr in G.neighbors(node)]
    return float(np.mean(vals)) if vals else 0.0

def pearson_r(x, y):
    x = np.asarray(x, float)
    y = np.asarray(y, float)
    if len(x) < 3:
        return float("nan")
    x = x - x.mean()
    y = y - y.mean()
    denom = (np.linalg.norm(x) * np.linalg.norm(y) + 1e-12)
    return float(np.dot(x, y) / denom)

def estimate_lightcone_speed_moment(cone_matrix, cone_times, p=1.0, floor=0.0):
    """
    Robust 'front speed' from the p-th radial moment of the signal.

    Let A(d,t) = max(|signal(d,t)| - floor, 0).
    Define effective radius:
        R(t) = ( Σ_d d^p A(d,t) / Σ_d A(d,t) )^(1/p)
    Fit R(t) = c*t + b.

    Returns (c_est, b_est, n_used).
    """
    M = np.asarray(cone_matrix, float)
    times = np.asarray(cone_times, float)
    if M.size == 0 or len(times) < 3:
        return float("nan"), float("nan"), 0

    A = np.abs(M)
    dgrid = np.arange(A.shape[1], dtype=float)

    Rs = []
    ts = []
    for ti in range(A.shape[0]):
        row = A[ti] - floor
        row = np.clip(row, 0.0, None)
        s = float(np.sum(row))
        if s <= 1e-12:
            continue
        moment = float(np.sum((dgrid ** p) * row) / s)
        R = moment ** (1.0 / p)
        Rs.append(R)
        ts.append(times[ti])

    if len(ts) < 3:
        return float("nan"), float("nan"), len(ts)

    ts = np.asarray(ts, float)
    Rs = np.asarray(Rs, float)

    Afit = np.vstack([ts, np.ones_like(ts)]).T
    c_est, b_est = np.linalg.lstsq(Afit, Rs, rcond=None)[0]
    return float(c_est), float(b_est), len(ts)


def estimate_lightcone_speed_global_threshold(cone_matrix, cone_times, noise_rows=5, sigma=3.0):
    """
    Estimate front speed using a global threshold:
    threshold = mean(|signal| in early rows) + sigma * std(|signal| in early rows)
    Then for each later time slice, front is farthest d with |signal| >= threshold.
    """
    M = np.asarray(cone_matrix, float)
    times = np.asarray(cone_times, float)
    if M.size == 0 or len(times) < max(3, noise_rows):
        return float("nan"), float("nan"), 0

    A = np.abs(M)

    # estimate noise from earliest rows
    early = A[:noise_rows, :]
    mu = float(np.mean(early))
    sd = float(np.std(early))
    thr = mu + sigma * sd

    ds, ts = [], []
    for ti in range(noise_rows, A.shape[0]):
        row = A[ti]
        idx = np.where(row >= thr)[0]
        if len(idx) == 0:
            continue
        ds.append(int(np.max(idx)))
        ts.append(times[ti])

    if len(ts) < 3:
        return float("nan"), float("nan"), len(ts)

    ts = np.asarray(ts, float)
    ds = np.asarray(ds, float)
    Afit = np.vstack([ts, np.ones_like(ts)]).T
    c_est, b_est = np.linalg.lstsq(Afit, ds, rcond=None)[0]
    return float(c_est), float(b_est), len(ts)


def estimate_lightcone_speed(cone_matrix, cone_times, frac=0.35):
    """
    Estimate front speed c from cone_matrix (Tbins x Dbins).
    For each time slice, find farthest d where |signal| >= frac * peak(signal at that time).
    Fit d = c*t + b.
    """
    M = np.asarray(cone_matrix, float)
    times = np.asarray(cone_times, float)
    if M.size == 0:
        return float("nan"), float("nan"), 0

    A = np.abs(M)
    ds, ts = [], []
    for ti in range(A.shape[0]):
        row = A[ti]
        peak = float(np.max(row))
        if peak <= 1e-12:
            continue
        thr = frac * peak
        idx = np.where(row >= thr)[0]
        if len(idx) == 0:
            continue
        ds.append(int(np.max(idx)))
        ts.append(times[ti])

    if len(ts) < 3:
        return float("nan"), float("nan"), len(ts)

    ts = np.asarray(ts, float)
    ds = np.asarray(ds, float)
    Afit = np.vstack([ts, np.ones_like(ts)]).T
    c_est, b_est = np.linalg.lstsq(Afit, ds, rcond=None)[0]
    return float(c_est), float(b_est), len(ts)

def estimate_corr_length(ds, corr, fit_min=2, fit_max=8):
    """Fit |corr(d)| ~ exp(-d/xi) using log-linear fit."""
    ds = np.asarray(ds, float)
    corr = np.asarray(corr, float)
    mask = (ds >= fit_min) & (ds <= fit_max) & (np.abs(corr) > 1e-6)
    x = ds[mask]
    y = np.log(np.abs(corr[mask]))
    if len(x) < 3:
        return float("nan")
    Afit = np.vstack([x, np.ones_like(x)]).T
    m, a = np.linalg.lstsq(Afit, y, rcond=None)[0]
    if m >= -1e-9:
        return float("nan")
    return float(-1.0 / m)

def nodes_within_radius(G, center, radius):
    dist = bfs_distances(G, center, radius)
    return list(dist.keys()), dist

def curvature_profile_vs_distance(G, source, max_d):
    dist_map = bfs_distances(G, source, max_d)
    buckets = [[] for _ in range(max_d + 1)]
    for v, d in dist_map.items():
        if d <= max_d:
            buckets[d].append(node_curvature_proxy(G, v))
    prof = np.zeros(max_d + 1, dtype=float)
    for d in range(max_d + 1):
        prof[d] = float(np.mean(buckets[d])) if buckets[d] else np.nan
    return prof

def density_profile_vs_distance(G, source, max_d):
    dist_map = bfs_distances(G, source, max_d)
    buckets = [[] for _ in range(max_d + 1)]
    for v, d in dist_map.items():
        if d <= max_d:
            buckets[d].append(abs(G.nodes[v]["psi"]) ** 2)
    prof = np.zeros(max_d + 1, dtype=float)
    for d in range(max_d + 1):
        prof[d] = float(np.mean(buckets[d])) if buckets[d] else np.nan
    return prof

def init_graph():
    G = nx.erdos_renyi_graph(n=N, p=INIT_P, seed=SEED)
    if G.number_of_edges() < N:
        for _ in range(N // 2):
            i = random.randrange(N)
            j = random.randrange(N)
            if i != j:
                G.add_edge(*edge_key(i, j))

    for i in G.nodes():
        re = np.random.normal(0.0, 0.6)
        im = np.random.normal(0.0, 0.6)
        G.nodes[i]["psi"] = re + 1j * im

    for a, b in G.edges():
        G[a][b]["theta"] = float(np.random.uniform(-math.pi, math.pi))
    return G

def metropolis_accept(dS: float, temp: float) -> bool:
    if dS <= 0:
        return True
    if temp <= 1e-12:
        return False
    return random.random() < math.exp(-dS / temp)

def psi_update(G: nx.Graph):
    i = random.randrange(N)
    e0 = local_energy_around_node_for_psi(G, i)
    psi0 = G.nodes[i]["psi"]

    dpsi = (np.random.normal(0.0, PSI_STEP) + 1j * np.random.normal(0.0, PSI_STEP))
    G.nodes[i]["psi"] = psi0 + dpsi

    e1 = local_energy_around_node_for_psi(G, i)
    dS = e1 - e0

    if metropolis_accept(dS, TEMP):
        return True
    G.nodes[i]["psi"] = psi0
    return False

def theta_update(G: nx.Graph):
    if G.number_of_edges() == 0:
        return False
    a, b = random.choice(list(G.edges()))
    e0 = local_energy_around_edge_for_theta(G, a, b)
    th0 = G[a][b]["theta"]

    G[a][b]["theta"] = wrap_angle(th0 + float(np.random.normal(0.0, THETA_STEP)))

    e1 = local_energy_around_edge_for_theta(G, a, b)
    dS = e1 - e0

    if metropolis_accept(dS, TEMP):
        return True
    G[a][b]["theta"] = th0
    return False

def rewire_move(G: nx.Graph):
    """
    Add/remove an edge with local ΔS using local_energy_for_rewire (includes stress term on edges).
    """
    i = random.randrange(N)
    do_add = (random.random() < 0.50)

    if do_add:
        j = propose_link_target(G, i)
        if j is None:
            return False
        a, b = edge_key(i, j)
        if G.has_edge(a, b):
            return False

        e0 = local_energy_for_rewire(G, i, j)

        G.add_edge(a, b)
        G[a][b]["theta"] = float(np.random.uniform(-math.pi, math.pi))

        e1 = local_energy_for_rewire(G, i, j)
        dS = e1 - e0

        if metropolis_accept(dS, TEMP):
            return True

        if G.has_edge(a, b):
            G.remove_edge(a, b)
        return False

    # remove
    if G.degree(i) == 0:
        return False
    j = random.choice(list(G.neighbors(i)))
    a, b = edge_key(i, j)
    if not G.has_edge(a, b):
        return False

    e0 = local_energy_for_rewire(G, i, j)
    th0 = G[a][b]["theta"]

    G.remove_edge(a, b)
    e1 = local_energy_for_rewire(G, i, j)
    dS = e1 - e0

    if metropolis_accept(dS, TEMP):
        return True

    # rollback
    G.add_edge(a, b)
    G[a][b]["theta"] = th0
    return False

def run(show_plot=True):
    G = init_graph()
    nodes = list(G.nodes())
    rng = np.random.default_rng(SEED + 123)

    print(f">> Relational Physics v2.1 | N={N} | init edges={G.number_of_edges()}")
    print(">> Equilibrating with rewiring...")

    eq_log = {"moves": [], "mean_deg": [], "tri_flux": [], "corr_d1": []}
    tri_samples = sample_triangles(G, max_tri=1200)

    for t in range(1, EQUIL_MOVES + 1):
        r = random.random()
        if r < P_PSI:
            psi_update(G)
        elif r < P_PSI + P_THETA:
            theta_update(G)
        else:
            rewire_move(G)

        if t % LOG_EVERY == 0 or t == EQUIL_MOVES:
            md = float(np.mean([G.degree(i) for i in nodes]))

            # Refresh triangle sample occasionally (rewiring changes triangle set)
            if (t % (10 * LOG_EVERY) == 0) or (t == LOG_EVERY) or (len(tri_samples) < 200):
                tri_samples = sample_triangles(G, max_tri=2500)

            tri_vals = []
            for tri in tri_samples:
                v = triangle_flux_real(G, tri)
                if not np.isnan(v):
                    tri_vals.append(v)
                if len(tri_vals) >= 250:
                    break
            tri_flux = float(np.mean(tri_vals)) if tri_vals else 0.0

            # normalized neighbor correlator (bounded ~[-1,1])
            if G.number_of_edges() > 0:
                es = random.sample(list(G.edges()), k=min(300, G.number_of_edges()))
                vals = []
                for a, b in es:
                    psi_a = G.nodes[a]["psi"]
                    psi_b = G.nodes[b]["psi"]
                    denom = (abs(psi_a) * abs(psi_b) + 1e-12)
                    vals.append(float(np.real(np.conjugate(psi_a) * U_ij(G, a, b) * psi_b) / denom))
                corr1 = float(np.mean(vals))
            else:
                corr1 = 0.0

            eq_log["moves"].append(t)
            eq_log["mean_deg"].append(md)
            eq_log["tri_flux"].append(tri_flux)
            eq_log["corr_d1"].append(corr1)

            print(f"  eq {t:6d}/{EQUIL_MOVES} | mean_deg={md:.2f} | mean Re(loop)={tri_flux:.3f} | d=1 corr={corr1:.3f}")

    print(">> Equilibration done.")
    tri_count_est = sum(nx.triangles(G).values()) // 3
    print(">> triangle count (exact):", tri_count_est)
    print(">> Running gauge invariance check...")

    # Gauge invariance check: resample live triangles now
    tri_samples = sample_triangles(G, max_tri=2500)

    def tri_is_alive(tri):
        i, j, k = tri
        return (G.has_edge(*edge_key(i, j)) and
                G.has_edge(*edge_key(j, k)) and
                G.has_edge(*edge_key(k, i)))

    tri_check = [tri for tri in tri_samples if tri_is_alive(tri)][:300]
    pair_check = [tuple(rng.choice(nodes, size=2, replace=False)) for _ in range(250)]

    tri_before = np.array([triangle_flux_real(G, tri) for tri in tri_check], dtype=float)
    pair_before = np.array([np.real(gauge_invariant_two_point(G, i, j)) for i, j in pair_check], dtype=float)

    apply_random_gauge_transform(G, rng)

    tri_after = np.array([triangle_flux_real(G, tri) for tri in tri_check], dtype=float)
    pair_after = np.array([np.real(gauge_invariant_two_point(G, i, j)) for i, j in pair_check], dtype=float)

    tri_diff = float(np.nanmax(np.abs(tri_after - tri_before))) if len(tri_before) else 0.0
    pair_diff = float(np.max(np.abs(pair_after - pair_before))) if len(pair_before) else 0.0

    print(f"  gauge check: max |Δ triangle flux| = {tri_diff:.3e}")
    print(f"  gauge check: max |Δ line correlator| = {pair_diff:.3e}")

    # Correlator vs distance
    print(">> Estimating gauge-invariant correlator vs hop distance...")
    ds, corr, corr_counts = correlator_vs_distance(
        G, samples=CORR_SAMPLES, max_d=MAX_DIST_CORR, seed=SEED + 999
    )

    xi_est = estimate_corr_length(ds, corr, fit_min=XI_FIT_MIN, fit_max=min(XI_FIT_MAX, MAX_DIST_CORR))

    # Curvature–density sample (baseline)
    comp = max(nx.connected_components(G), key=len)
    sample_nodes = random.sample(list(comp), k=min(CURV_SAMPLE, len(comp)))
    rho = np.array([abs(G.nodes[i]["psi"]) ** 2 for i in sample_nodes], dtype=float)
    curv = np.array([node_curvature_proxy(G, i) for i in sample_nodes], dtype=float)
    r_rho_curv = pearson_r(rho, curv)

    # -----------------------
    # QUENCH EXPERIMENT
    # -----------------------
    quench_src = None
    delta_curv_prof = None
    delta_rho_prof = None

    if DO_QUENCH:
        print(">> Running local quench (energy injection) and measuring Δcurvature profile...")

        comp_list = list(comp)
        quench_src = random.choice(comp_list)

        curv_before = curvature_profile_vs_distance(G, quench_src, MAX_DIST_QUENCH)
        rho_before = density_profile_vs_distance(G, quench_src, MAX_DIST_QUENCH)

        region_nodes, _ = nodes_within_radius(G, quench_src, QUENCH_RADIUS)

        for v in region_nodes:
            G.nodes[v]["psi"] += (QUENCH_STRENGTH + 0.0j)

        # Relax after quench. If you want curvature response, allow rewiring here.
        for _ in range(QUENCH_RELAX_MOVES):
            rr = random.random()
            if QUENCH_ALLOW_REWIRE and rr < QUENCH_P_REWIRE:
                rewire_move(G)
            else:
                # psi/theta with same relative proportions as base (excluding rewire)
                if rr < (P_PSI / (P_PSI + P_THETA)):
                    psi_update(G)
                else:
                    theta_update(G)

        curv_after = curvature_profile_vs_distance(G, quench_src, MAX_DIST_QUENCH)
        rho_after = density_profile_vs_distance(G, quench_src, MAX_DIST_QUENCH)

        delta_curv_prof = curv_after - curv_before
        delta_rho_prof = rho_after - rho_before

        print("   d : Δrho(d), Δcurv(d)")
        for d in range(min(MAX_DIST_QUENCH + 1, 10)):
            print(f"  {d:2d}: {delta_rho_prof[d]: .3f}, {delta_curv_prof[d]: .3f}")

    # -----------------------
    # LIGHT CONE EXPERIMENT (geometry frozen by default)
    # -----------------------
    print(">> Light cone experiment (geometry frozen)...")
    src = random.choice(list(comp))

    dist_map = bfs_distances(G, src, MAX_DIST_LIGHTCONE)
    baseline_abspsi = {i: abs(G.nodes[i]["psi"]) for i in G.nodes()}

    G.nodes[src]["psi"] += (PERTURB_EPS + 0.0j)

    cone_times = []
    cone_matrix = []

    for t in range(1, PROP_MOVES + 1):
        rr = random.random()
        if rr < (P_PSI / (P_PSI + P_THETA)):
            psi_update(G)
        else:
            theta_update(G)

        if t % LOG_EVERY == 0 or t == PROP_MOVES:
            cone_times.append(t)
            cone_matrix.append(lightcone_measurement(dist_map, baseline_abspsi, G, MAX_DIST_LIGHTCONE))

    cone_matrix = np.array(cone_matrix)
    A = np.abs(cone_matrix)
    dgrid = np.arange(A.shape[1], dtype=float)

    Rs, ts_used = [], []
    for ti, t in enumerate(cone_times):
        row = A[ti]
        s = float(np.sum(row))
        if s <= 1e-12:
            continue
        R = math.sqrt(float(np.sum((dgrid**2)*row)/s))  # p=2 radius
        Rs.append(R)
        ts_used.append(t)

    fit = fit_front_scaling(Rs, ts_used)
    print(f">> Front scaling fit: linear SSE={fit['lin_sse']:.3g}, sqrt SSE={fit['sqrt_sse']:.3g}")
    print(f"   linear:  R≈{fit['lin_a']:.3e} t + {fit['lin_b']:.3f}")
    print(f"   sqrt:    R≈{fit['sqrt_a']:.3e} sqrt(t) + {fit['sqrt_b']:.3f}")
    D_eff = fit["sqrt_a"]**2
    print(f">> Diffusion proxy: D ≈ {D_eff:.3e} hop^2/move")

    r_resp = float("nan")
    if DO_QUENCH and (delta_rho_prof is not None) and (delta_curv_prof is not None):
        mask = ~np.isnan(delta_rho_prof) & ~np.isnan(delta_curv_prof)
        if np.any(mask):
            r_resp = pearson_r(delta_rho_prof[mask], delta_curv_prof[mask])
        print(f">> Quench response coupling r(Δρ,Δcurv) ≈ {r_resp:.3f}")



    c_est, b_est, npts = estimate_lightcone_speed_moment(cone_matrix, cone_times, p=2.0, floor=0.0)

    print(">> Effective numbers:")
    print(f"   correlation length xi ≈ {xi_est:.3f} (hop units)")
    print(f"ballistic fit slope v_lin ≈ {fit['lin_a']:.6e} hops/move (for comparison)")
    print(f"diffusive fit slope a_sqrt ≈ {fit['sqrt_a']:.6e} hops/sqrt(move)")
    print(f"diffusion proxy D ≈ {fit['sqrt_a']**2:.3e} hop^2/move")


    print(f"   curvature-density Pearson r ≈ {r_rho_curv:.3f}")


    metrics = {
        "xi": float(xi_est),
        "v_lin": float(fit["lin_a"]),
        "a_sqrt": float(fit["sqrt_a"]),
        "D": float(fit["sqrt_a"]**2),
        "lin_sse": float(fit["lin_sse"]),
        "sqrt_sse": float(fit["sqrt_sse"]),
        "r_rho_curv": float(r_rho_curv),
        "r_resp": float(r_resp) if "r_resp" in locals() else float("nan"),
        "mean_deg": float(np.mean([G.degree(i) for i in G.nodes()])),
        "tri_diff": float(tri_diff),
        "pair_diff": float(pair_diff),
    }

    if show_plot==False:
        return metrics

    # -----------------------
    # DASHBOARD
    # -----------------------
    fig, axs = plt.subplots(2, 3, figsize=(18, 10))
    plt.subplots_adjust(hspace=0.28, wspace=0.22)

    # (1) Equilibration diagnostics
    axs[0, 0].plot(eq_log["moves"], eq_log["mean_deg"], label="Mean degree")
    axs[0, 0].plot(eq_log["moves"], eq_log["tri_flux"], label="Mean Re(triangle loop)")
    axs[0, 0].plot(eq_log["moves"], eq_log["corr_d1"], label="Mean d=1 corr (Re, norm)")
    axs[0, 0].set_title("Equilibration diagnostics")
    axs[0, 0].set_xlabel("Moves")
    axs[0, 0].grid(True, alpha=0.3)
    axs[0, 0].legend()

    # (2) Correlator vs distance
    axs[0, 1].plot(ds, corr, marker="o")
    axs[0, 1].set_title("Gauge-invariant correlator vs hop distance (normalized)")
    axs[0, 1].set_xlabel("Hop distance d")
    axs[0, 1].set_ylabel("Mean Re( conj(psi_i) U_path psi_j )")
    axs[0, 1].grid(True, alpha=0.3)
    for d, c in zip(ds, corr_counts):
        if d in (1, 2, 3, 4, 6, 8, 10, 12) and c > 0:
            axs[0, 1].annotate(str(int(c)), (d, corr[d]), textcoords="offset points", xytext=(4, 4), fontsize=8)

    # (3) Quench response
    axs[0, 2].set_title("Quench response: Δρ(d) and Δcurvature(d)")
    axs[0, 2].set_xlabel("Hop distance d")
    axs[0, 2].grid(True, alpha=0.3)
    if DO_QUENCH and (delta_curv_prof is not None) and (delta_rho_prof is not None):
        dgrid = np.arange(len(delta_curv_prof))
        axs[0, 2].plot(dgrid, delta_rho_prof, marker="o", label="Δρ(d)=Δ|psi|^2")
        axs[0, 2].plot(dgrid, delta_curv_prof, marker="o", label="Δcurv(d)")
        axs[0, 2].legend()
        axs[0, 2].annotate(
            f"src={quench_src}, R={QUENCH_RADIUS}, rewire={QUENCH_ALLOW_REWIRE}",
            (0.02, 0.95),
            xycoords="axes fraction",
            va="top",
            fontsize=9,
        )
    else:
        axs[0, 2].text(0.5, 0.5, "Quench disabled", ha="center", va="center")

    # (4) Light cone heatmap
    im = axs[1, 0].imshow(
        cone_matrix,
        aspect="auto",
        origin="lower",
        extent=[0, MAX_DIST_LIGHTCONE, cone_times[0], cone_times[-1]],
    )
    axs[1, 0].set_title("Light cone probe: Δ|psi| vs distance/time (geometry frozen)")
    axs[1, 0].set_xlabel("Hop distance from source")
    axs[1, 0].set_ylabel("Moves (time)")
    plt.colorbar(im, ax=axs[1, 0], label="Mean Δ|psi|")

    # (5) Curvature vs density scatter
    axs[1, 1].scatter(rho, curv, s=18, alpha=0.7)

    axs[1, 2].axis("off")

    fig.suptitle(
        f"Relational Physics v2.1 | N={N} | κ={KAPPA}, β={BETA}, T={TEMP}, λE={LAMBDA_E}, μ={MU_DEG2}, λG={LAMBDA_G}\n"
        f"Gauge check: maxΔloop={tri_diff:.2e}, maxΔcorr={pair_diff:.2e} | xi≈{xi_est:.2f} | c≈{c_est:.3f} | r≈{r_rho_curv:.2f} | source={src}",
        fontsize=12,
    )



    plt.show()

    return rows
